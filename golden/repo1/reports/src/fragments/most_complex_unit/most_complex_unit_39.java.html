    public static boolean matchesAnyPattern(String line, List<String> patterns) {
        for (String patternString : patterns) {
            try {
                if (Pattern.compile(patternString).matcher(line).matches()) {
                    return true;
                }
            } catch (PatternSyntaxException e) {
                LOG.debug(e);
            }
        }
        return false;
    }
