    protected String getMessage(String message, JsonLocation jsonLocation) {
        String[] lines = message.split("\n");
        if (jsonLocation != null && lines.length > 0) {
            String line = lines[0];
            line += " (line " + jsonLocation.getLineNr();
            line += (", column " + jsonLocation.getColumnNr() + ")");
            return line;
        }
        return message;
    }
