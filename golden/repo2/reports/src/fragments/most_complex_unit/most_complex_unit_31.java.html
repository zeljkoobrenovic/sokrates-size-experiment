    public boolean equals(final Object o) {
        if (this == o) {
            return true;
        }
        if (o == null || getClass() != o.getClass()) {
            return false;
        }
        final DirectoryNode that = (DirectoryNode) o;
        return Objects.equals(this.value, that.value)
                && Objects.equals(this.parent, that.parent);
    }
