    public File getFileProperty(String propertyName) {
        String propertyValue = properties.getProperty(propertyName);
        if (propertyValue != null) {
            File file = new File(propertyValue);
            if (file.exists()) {
                return file;
            } else {
                return null;
            }
        }
        return null;
    }
