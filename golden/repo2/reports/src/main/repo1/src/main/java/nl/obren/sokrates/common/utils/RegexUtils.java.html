private static final Log LOG = LogFactory.getLog(RegexUtils.class); private static Map<String, Pattern> compiledPatterns = new HashMap<>(); public static boolean matchesEntirely(String regexPattern, String content) { try { String key = regexPattern + " ::->:: " + content;
